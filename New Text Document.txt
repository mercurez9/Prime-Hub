--[[
	NovaUI v1.0.0
	A compact, dark-navy Roblox UI library.
	Window -> Tab -> Section -> Elements

	Usage:
		local Nova = loadstring(game:HttpGet("https://raw.githubusercontent.com/<you>/<repo>/main/NovaUI.lua"))()
		local Window = Nova:CreateWindow({ Title = "Nova | v1.0" })
]]

local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")

local Nova = {
	Version = "1.0.0",
	Flags = {},   -- Nova.Flags["FlagName"] = current value
	Options = {}, -- Nova.Options["FlagName"] = element object (:Set / :Get)
}

Nova.Theme = {
	Background = Color3.fromRGB(14, 16, 24),
	Panel = Color3.fromRGB(21, 24, 35),
	Element = Color3.fromRGB(29, 33, 47),
	ElementHover = Color3.fromRGB(37, 42, 60),
	Stroke = Color3.fromRGB(58, 74, 180),
	Accent = Color3.fromRGB(72, 102, 255),
	Title = Color3.fromRGB(125, 150, 255),
	Text = Color3.fromRGB(236, 239, 250),
	SubText = Color3.fromRGB(140, 148, 178),
	Off = Color3.fromRGB(52, 58, 80),
	Font = Enum.Font.GothamMedium,
	FontBold = Enum.Font.GothamBold,
}

-- ───────────────────────── helpers ─────────────────────────

local function New(class, props, children)
	local inst = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then inst[k] = v end
	end
	for _, c in ipairs(children or {}) do c.Parent = inst end
	if props and props.Parent then inst.Parent = props.Parent end
	return inst
end

local function Tween(inst, props, t, style, dir)
	local tw = TweenService:Create(
		inst,
		TweenInfo.new(t or 0.18, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	tw:Play()
	return tw
end

local function Corner(p, r)
	return New("UICorner", { CornerRadius = UDim.new(0, r), Parent = p })
end

local function Stroke(p, color, thickness, transparency)
	return New("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Parent = p,
	})
end

local function Pad(p, l, t, r, b)
	return New("UIPadding", {
		PaddingLeft = UDim.new(0, l), PaddingTop = UDim.new(0, t),
		PaddingRight = UDim.new(0, r or l), PaddingBottom = UDim.new(0, b or t),
		Parent = p,
	})
end

local function List(p, padding)
	return New("UIListLayout", {
		Padding = UDim.new(0, padding or 6),
		SortOrder = Enum.SortOrder.LayoutOrder,
		Parent = p,
	})
end

local function safe(fn, ...)
	if type(fn) ~= "function" then return end
	local ok, err = pcall(fn, ...)
	if not ok then warn("[NovaUI] callback error: " .. tostring(err)) end
end

local function isPointer(i)
	return i.UserInputType == Enum.UserInputType.MouseButton1
		or i.UserInputType == Enum.UserInputType.Touch
end

local function isMove(i)
	return i.UserInputType == Enum.UserInputType.MouseMovement
		or i.UserInputType == Enum.UserInputType.Touch
end

-- ───────────────────────── window ─────────────────────────

function Nova:CreateWindow(opts)
	opts = opts or {}

	local T = table.clone(Nova.Theme)
	for k, v in pairs(opts.Theme or {}) do T[k] = v end

	local Window = { Tabs = {}, Visible = true, Minimized = false }
	local conns = {}

	local function bind(signal, fn)
		local c = signal:Connect(fn)
		table.insert(conns, c)
		return c
	end

	local function Lbl(p)
		p.BackgroundTransparency = 1
		p.Font = p.Font or T.Font
		p.TextSize = p.TextSize or 13
		p.TextColor3 = p.TextColor3 or T.Text
		p.TextXAlignment = p.TextXAlignment or Enum.TextXAlignment.Left
		return New("TextLabel", p)
	end

	local function Hover(row, hit)
		hit.MouseEnter:Connect(function() Tween(row, { BackgroundColor3 = T.ElementHover }, 0.12) end)
		hit.MouseLeave:Connect(function() Tween(row, { BackgroundColor3 = T.Element }, 0.12) end)
	end

	local function reg(o, obj)
		if o.Flag then Nova.Options[o.Flag] = obj end
		return obj
	end

	-- ScreenGui
	local gui = New("ScreenGui", {
		Name = "NovaUI",
		ResetOnSpawn = false,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
		DisplayOrder = 999,
	})
	local ok = pcall(function() gui.Parent = (gethui and gethui()) or CoreGui end)
	if not ok or not gui.Parent then
		gui.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")
	end

	-- main frame
	local size = opts.Size or UDim2.fromOffset(560, 380)
	local main = New("Frame", {
		Name = "Main",
		Size = size,
		Position = UDim2.new(0.5, -size.X.Offset / 2, 0.5, -size.Y.Offset / 2),
		BackgroundColor3 = T.Background,
		BorderSizePixel = 0,
		Parent = gui,
	})
	Corner(main, 8)
	Stroke(main, T.Stroke, 1.5)

	-- title bar
	local top = New("Frame", {
		Size = UDim2.new(1, 0, 0, 34), BackgroundColor3 = T.Panel,
		BorderSizePixel = 0, Parent = main,
	})
	Corner(top, 8)
	New("Frame", { -- squares off the bottom corners of the title bar
		Size = UDim2.new(1, 0, 0, 8), Position = UDim2.new(0, 0, 1, -8),
		BackgroundColor3 = T.Panel, BorderSizePixel = 0, Parent = top,
	})
	local titleLbl = Lbl({
		Size = UDim2.new(1, -90, 1, 0), Position = UDim2.fromOffset(12, 0),
		Text = opts.Title or "Nova UI", TextColor3 = T.Title,
		Font = T.FontBold, TextSize = 14, Parent = top,
	})

	local function TopButton(text, xOffset)
		local b = New("TextButton", {
			Size = UDim2.fromOffset(24, 24), Position = UDim2.new(1, xOffset, 0.5, -12),
			BackgroundColor3 = T.Element, BackgroundTransparency = 1,
			Text = text, TextColor3 = T.SubText, Font = T.FontBold, TextSize = 16,
			AutoButtonColor = false, Parent = top,
		})
		Corner(b, 5)
		b.MouseEnter:Connect(function()
			Tween(b, { BackgroundTransparency = 0, TextColor3 = T.Text }, 0.12)
		end)
		b.MouseLeave:Connect(function()
			Tween(b, { BackgroundTransparency = 1, TextColor3 = T.SubText }, 0.12)
		end)
		return b
	end
	local minBtn = TopButton("–", -58)
	local closeBtn = TopButton("×", -30)

	-- body
	local body = New("Frame", {
		Size = UDim2.new(1, 0, 1, -34), Position = UDim2.fromOffset(0, 34),
		BackgroundTransparency = 1, Parent = main,
	})
	local sidebar = New("Frame", {
		Size = UDim2.new(0, 124, 1, -12), Position = UDim2.fromOffset(8, 4),
		BackgroundColor3 = T.Panel, BorderSizePixel = 0, Parent = body,
	})
	Corner(sidebar, 6)
	Stroke(sidebar, T.Stroke, 1, 0.7)
	Pad(sidebar, 6, 6)
	List(sidebar, 4)

	local pages = New("Frame", {
		Size = UDim2.new(1, -148, 1, -12), Position = UDim2.fromOffset(140, 4),
		BackgroundTransparency = 1, Parent = body,
	})

	-- resize grip (bottom-right)
	local grip = New("TextButton", {
		Size = UDim2.fromOffset(16, 16), Position = UDim2.new(1, -16, 1, -16),
		BackgroundTransparency = 1, Text = "◢", TextColor3 = T.Stroke,
		TextSize = 12, Font = T.Font, AutoButtonColor = false, ZIndex = 5, Parent = main,
	})

	-- dragging (title bar)
	do
		local dragging, start, startPos
		bind(top.InputBegan, function(i)
			if isPointer(i) then
				dragging, start, startPos = true, i.Position, main.Position
				i.Changed:Connect(function()
					if i.UserInputState == Enum.UserInputState.End then dragging = false end
				end)
			end
		end)
		bind(UserInputService.InputChanged, function(i)
			if dragging and isMove(i) then
				local d = i.Position - start
				main.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + d.X,
					startPos.Y.Scale, startPos.Y.Offset + d.Y
				)
			end
		end)
	end

	-- resizing (grip)
	do
		local resizing, rStart, rSize
		local minW, minH = opts.MinSize and opts.MinSize.X or 420, opts.MinSize and opts.MinSize.Y or 280
		bind(grip.InputBegan, function(i)
			if isPointer(i) then
				resizing, rStart, rSize = true, i.Position, main.AbsoluteSize
				i.Changed:Connect(function()
					if i.UserInputState == Enum.UserInputState.End then resizing = false end
				end)
			end
		end)
		bind(UserInputService.InputChanged, function(i)
			if resizing and isMove(i) then
				local d = i.Position - rStart
				main.Size = UDim2.fromOffset(math.max(minW, rSize.X + d.X), math.max(minH, rSize.Y + d.Y))
			end
		end)
	end

	-- minimize / close
	local restoreSize
	minBtn.MouseButton1Click:Connect(function()
		Window.Minimized = not Window.Minimized
		if Window.Minimized then
			restoreSize = main.Size
			body.Visible, grip.Visible = false, false
			Tween(main, { Size = UDim2.fromOffset(main.AbsoluteSize.X, 34) }, 0.2)
		else
			Tween(main, { Size = restoreSize }, 0.2).Completed:Connect(function()
				if not Window.Minimized then body.Visible, grip.Visible = true, true end
			end)
		end
	end)
	closeBtn.MouseButton1Click:Connect(function() Window:Destroy() end)

	-- toggle key
	local toggleKey = opts.ToggleKey or Enum.KeyCode.RightShift
	bind(UserInputService.InputBegan, function(i, gp)
		if not gp and i.KeyCode == toggleKey then Window:Toggle() end
	end)

	-- notifications
	local notifHolder = New("Frame", {
		Size = UDim2.new(0, 260, 1, -20), Position = UDim2.new(1, -270, 0, 10),
		BackgroundTransparency = 1, Parent = gui,
	})
	New("UIListLayout", {
		Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder,
		VerticalAlignment = Enum.VerticalAlignment.Bottom, Parent = notifHolder,
	})

	-- ───────────── window methods ─────────────

	function Window:SetTitle(text) titleLbl.Text = text end
	function Window:SetVisible(v) self.Visible = v; main.Visible = v end
	function Window:Toggle() self:SetVisible(not self.Visible) end

	function Window:Destroy()
		for _, c in ipairs(conns) do c:Disconnect() end
		gui:Destroy()
	end

	function Window:Notify(n)
		n = n or {}
		local duration = n.Duration or 4
		local card = New("CanvasGroup", {
			Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
			BackgroundColor3 = T.Panel, GroupTransparency = 1, Parent = notifHolder,
		})
		Corner(card, 6)
		Stroke(card, T.Stroke, 1.2)
		Pad(card, 10, 8)
		List(card, 4)
		Lbl({
			Size = UDim2.new(1, 0, 0, 16), Text = n.Title or "Notification",
			TextColor3 = T.Title, Font = T.FontBold, LayoutOrder = 1, Parent = card,
		})
		Lbl({
			Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
			Text = n.Content or "", TextColor3 = T.SubText, TextSize = 12,
			TextWrapped = true, LayoutOrder = 2, Parent = card,
		})
		local track = New("Frame", {
			Size = UDim2.new(1, 0, 0, 3), BackgroundColor3 = T.Off,
			BorderSizePixel = 0, LayoutOrder = 3, Parent = card,
		})
		Corner(track, 2)
		local fill = New("Frame", {
			Size = UDim2.fromScale(1, 1), BackgroundColor3 = T.Accent,
			BorderSizePixel = 0, Parent = track,
		})
		Corner(fill, 2)

		Tween(card, { GroupTransparency = 0 }, 0.25)
		Tween(fill, { Size = UDim2.fromScale(0, 1) }, duration, Enum.EasingStyle.Linear)
		task.delay(duration, function()
			if card.Parent then
				Tween(card, { GroupTransparency = 1 }, 0.25).Completed:Connect(function() card:Destroy() end)
			end
		end)
	end

	-- ───────────── tabs ─────────────

	local current
	local function selectTab(tab)
		if current == tab then return end
		if current then current._deselect() end
		current = tab
		tab._select()
	end
	function Window:SelectTab(tab) selectTab(tab) end

	function Window:CreateTab(name)
		local Tab = { Name = name }
		table.insert(self.Tabs, Tab)

		local btn = New("TextButton", {
			Size = UDim2.new(1, 0, 0, 30), BackgroundColor3 = T.Element,
			BackgroundTransparency = 1, Text = "", AutoButtonColor = false,
			LayoutOrder = #self.Tabs, Parent = sidebar,
		})
		Corner(btn, 6)
		local bar = New("Frame", {
			Size = UDim2.fromOffset(3, 14), Position = UDim2.new(0, 2, 0.5, -7),
			BackgroundColor3 = T.Accent, BackgroundTransparency = 1,
			BorderSizePixel = 0, Parent = btn,
		})
		Corner(bar, 2)
		local lbl = Lbl({
			Size = UDim2.new(1, -16, 1, 0), Position = UDim2.fromOffset(12, 0),
			Text = name, TextColor3 = T.SubText, Parent = btn,
		})

		local page = New("ScrollingFrame", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0,
			ScrollBarThickness = 3, ScrollBarImageColor3 = T.Stroke,
			AutomaticCanvasSize = Enum.AutomaticSize.Y, CanvasSize = UDim2.new(),
			Visible = false, Parent = pages,
		})
		List(page, 8)
		Pad(page, 0, 0, 6, 0)

		Tab._select = function()
			page.Visible = true
			Tween(btn, { BackgroundTransparency = 0 }, 0.15)
			Tween(bar, { BackgroundTransparency = 0 }, 0.15)
			Tween(lbl, { TextColor3 = T.Text }, 0.15)
		end
		Tab._deselect = function()
			page.Visible = false
			Tween(btn, { BackgroundTransparency = 1 }, 0.15)
			Tween(bar, { BackgroundTransparency = 1 }, 0.15)
			Tween(lbl, { TextColor3 = T.SubText }, 0.15)
		end
		btn.MouseButton1Click:Connect(function() selectTab(Tab) end)
		if #self.Tabs == 1 then selectTab(Tab) end

		-- ───────────── sections ─────────────

		function Tab:CreateSection(sectionName)
			local Section = {}
			local order = 0
			local function nextOrder() order += 1; return order end

			local frame = New("Frame", {
				Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
				BackgroundColor3 = T.Panel, BorderSizePixel = 0, Parent = page,
			})
			Corner(frame, 6)
			Stroke(frame, T.Stroke, 1, 0.5)
			Pad(frame, 8, 8)
			List(frame, 6)
			Lbl({
				Size = UDim2.new(1, 0, 0, 16), Text = string.upper(sectionName or "Section"),
				TextColor3 = T.Title, Font = T.FontBold, TextSize = 11, LayoutOrder = 0, Parent = frame,
			})

			local function Row(h)
				local r = New("Frame", {
					Size = UDim2.new(1, 0, 0, h or 32), BackgroundColor3 = T.Element,
					BorderSizePixel = 0, ClipsDescendants = true,
					LayoutOrder = nextOrder(), Parent = frame,
				})
				Corner(r, 6)
				Stroke(r, T.Stroke, 1, 0.75)
				return r
			end

			-- Label ---------------------------------------------------
			function Section:AddLabel(text)
				local l = Lbl({
					Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
					Text = text or "", TextColor3 = T.SubText, TextSize = 12,
					TextWrapped = true, LayoutOrder = nextOrder(), Parent = frame,
				})
				local obj = {}
				function obj:Set(t) l.Text = t end
				return obj
			end

			-- Divider -------------------------------------------------
			function Section:AddDivider()
				New("Frame", {
					Size = UDim2.new(1, 0, 0, 1), BackgroundColor3 = T.Stroke,
					BackgroundTransparency = 0.6, BorderSizePixel = 0,
					LayoutOrder = nextOrder(), Parent = frame,
				})
			end

			-- Button --------------------------------------------------
			function Section:AddButton(o)
				o = o or {}
				local row = Row(32)
				local hit = New("TextButton", {
					Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1,
					Text = o.Name or "Button", TextColor3 = T.Text,
					Font = T.Font, TextSize = 13, AutoButtonColor = false, Parent = row,
				})
				Hover(row, hit)
				hit.MouseButton1Click:Connect(function()
					row.BackgroundColor3 = T.Accent
					Tween(row, { BackgroundColor3 = T.ElementHover }, 0.3)
					safe(o.Callback)
				end)
				local obj = {}
				function obj:SetText(t) hit.Text = t end
				return obj
			end

			-- Toggle --------------------------------------------------
			function Section:AddToggle(o)
				o = o or {}
				local row = Row(32)
				Lbl({
					Size = UDim2.new(1, -60, 1, 0), Position = UDim2.fromOffset(10, 0),
					Text = o.Name or "Toggle", Parent = row,
				})
				local sw = New("Frame", {
					Size = UDim2.fromOffset(38, 18), Position = UDim2.new(1, -48, 0.5, -9),
					BackgroundColor3 = T.Off, BorderSizePixel = 0, Parent = row,
				})
				Corner(sw, 9)
				local knob = New("Frame", {
					Size = UDim2.fromOffset(14, 14), Position = UDim2.fromOffset(2, 2),
					BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = sw,
				})
				Corner(knob, 7)
				local hit = New("TextButton", {
					Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", Parent = row,
				})
				Hover(row, hit)

				local obj = { Value = false }
				function obj:Set(v, silent)
					self.Value = v and true or false
					Tween(sw, { BackgroundColor3 = self.Value and T.Accent or T.Off }, 0.15)
					Tween(knob, { Position = self.Value and UDim2.fromOffset(22, 2) or UDim2.fromOffset(2, 2) }, 0.15)
					if o.Flag then Nova.Flags[o.Flag] = self.Value end
					if not silent then safe(o.Callback, self.Value) end
				end
				function obj:Get() return self.Value end
				hit.MouseButton1Click:Connect(function() obj:Set(not obj.Value) end)
				obj:Set(o.Default or false, true)
				return reg(o, obj)
			end

			-- Slider --------------------------------------------------
			function Section:AddSlider(o)
				o = o or {}
				local min, max, inc = o.Min or 0, o.Max or 100, o.Increment or 1
				local row = Row(46)
				Lbl({
					Size = UDim2.new(1, -80, 0, 18), Position = UDim2.fromOffset(10, 5),
					Text = o.Name or "Slider", Parent = row,
				})
				local valLbl = Lbl({
					Size = UDim2.new(0, 70, 0, 18), Position = UDim2.new(1, -80, 0, 5),
					TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = T.SubText, Parent = row,
				})
				local bar = New("Frame", {
					Size = UDim2.new(1, -20, 0, 6), Position = UDim2.fromOffset(10, 31),
					BackgroundColor3 = T.Off, BorderSizePixel = 0, Parent = row,
				})
				Corner(bar, 3)
				local fill = New("Frame", {
					Size = UDim2.fromScale(0, 1), BackgroundColor3 = T.Accent,
					BorderSizePixel = 0, Parent = bar,
				})
				Corner(fill, 3)
				local knob = New("Frame", {
					Size = UDim2.fromOffset(12, 12), AnchorPoint = Vector2.new(0.5, 0.5),
					Position = UDim2.fromScale(1, 0.5), BackgroundColor3 = Color3.new(1, 1, 1),
					BorderSizePixel = 0, Parent = fill,
				})
				Corner(knob, 6)
				local hit = New("TextButton", {
					Size = UDim2.new(1, 0, 0, 24), Position = UDim2.fromOffset(0, 22),
					BackgroundTransparency = 1, Text = "", Parent = row,
				})

				local obj = { Value = min }
				function obj:Set(v, silent)
					v = math.clamp(tonumber(v) or min, min, max)
					v = math.floor((v - min) / inc + 0.5) * inc + min
					v = math.clamp(tonumber(string.format("%.4f", v)), min, max)
					self.Value = v
					valLbl.Text = tostring(v) .. (o.Suffix or "")
					Tween(fill, { Size = UDim2.new((v - min) / (max - min), 0, 1, 0) }, 0.06)
					if o.Flag then Nova.Flags[o.Flag] = v end
					if not silent then safe(o.Callback, v) end
				end
				function obj:Get() return self.Value end

				local dragging = false
				local function update(x)
					local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
					obj:Set(min + (max - min) * rel)
				end
				hit.InputBegan:Connect(function(i)
					if isPointer(i) then dragging = true; update(i.Position.X) end
				end)
				bind(UserInputService.InputChanged, function(i)
					if dragging and isMove(i) then update(i.Position.X) end
				end)
				bind(UserInputService.InputEnded, function(i)
					if isPointer(i) then dragging = false end
				end)

				obj:Set(o.Default or min, true)
				return reg(o, obj)
			end

			-- Dropdown ------------------------------------------------
			function Section:AddDropdown(o)
				o = o or {}
				local options = o.Options or {}
				local row = Row(32)
				Lbl({
					Size = UDim2.new(0.5, -10, 0, 32), Position = UDim2.fromOffset(10, 0),
					Text = o.Name or "Dropdown", Parent = row,
				})
				local valLbl = Lbl({
					Size = UDim2.new(0.5, -30, 0, 32), Position = UDim2.new(0.5, 0, 0, 0),
					Text = "None", TextXAlignment = Enum.TextXAlignment.Right,
					TextColor3 = T.SubText, Parent = row,
				})
				local arrow = Lbl({
					Size = UDim2.fromOffset(16, 32), Position = UDim2.new(1, -24, 0, 0),
					Text = "▾", TextXAlignment = Enum.TextXAlignment.Center,
					TextColor3 = T.SubText, TextSize = 14, Parent = row,
				})
				local hit = New("TextButton", {
					Size = UDim2.new(1, 0, 0, 32), BackgroundTransparency = 1, Text = "", Parent = row,
				})
				local list = New("ScrollingFrame", {
					Position = UDim2.fromOffset(6, 36), Size = UDim2.new(1, -12, 0, 0),
					BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 2,
					ScrollBarImageColor3 = T.Stroke, AutomaticCanvasSize = Enum.AutomaticSize.Y,
					CanvasSize = UDim2.new(), Parent = row,
				})
				List(list, 2)
				Hover(row, hit)

				local obj = { Value = nil, Options = options }
				local buttons, open = {}, false

				local function listH()
					local n = #options
					return math.min(n * 24 + math.max(n - 1, 0) * 2, 120)
				end
				local function setOpen(v)
					open = v
					Tween(row, { Size = UDim2.new(1, 0, 0, v and (32 + listH() + 10) or 32) }, 0.18)
					Tween(arrow, { Rotation = v and 180 or 0 }, 0.18)
				end
				local function paint()
					for key, b in pairs(buttons) do
						b.TextColor3 = (key == tostring(obj.Value)) and T.Accent or T.Text
					end
				end
				local function build()
					for _, b in pairs(buttons) do b:Destroy() end
					buttons = {}
					for idx, opt in ipairs(options) do
						local b = New("TextButton", {
							Size = UDim2.new(1, -4, 0, 24), BackgroundColor3 = T.Panel,
							Text = tostring(opt), Font = T.Font, TextSize = 12, TextColor3 = T.Text,
							AutoButtonColor = false, LayoutOrder = idx, Parent = list,
						})
						Corner(b, 4)
						b.MouseButton1Click:Connect(function() obj:Set(opt); setOpen(false) end)
						buttons[tostring(opt)] = b
					end
					list.Size = UDim2.new(1, -12, 0, listH())
					paint()
					if open then setOpen(true) end
				end

				function obj:Set(v, silent)
					self.Value = v
					valLbl.Text = v ~= nil and tostring(v) or "None"
					paint()
					if o.Flag then Nova.Flags[o.Flag] = v end
					if not silent then safe(o.Callback, v) end
				end
				function obj:Get() return self.Value end
				function obj:Refresh(newOptions, keepValue)
					options = newOptions
					self.Options = newOptions
					if not (keepValue and table.find(options, self.Value)) then self:Set(nil, true) end
					build()
				end

				hit.MouseButton1Click:Connect(function() setOpen(not open) end)
				build()
				if o.Default ~= nil then obj:Set(o.Default, true) end
				return reg(o, obj)
			end

			-- Textbox -------------------------------------------------
			function Section:AddTextbox(o)
				o = o or {}
				local row = Row(32)
				Lbl({
					Size = UDim2.new(0.45, -10, 1, 0), Position = UDim2.fromOffset(10, 0),
					Text = o.Name or "Textbox", Parent = row,
				})
				local box = New("TextBox", {
					Size = UDim2.new(0.55, -14, 0, 22), AnchorPoint = Vector2.new(1, 0.5),
					Position = UDim2.new(1, -6, 0.5, 0), BackgroundColor3 = T.Panel,
					Text = o.Default or "", PlaceholderText = o.Placeholder or "Type here...",
					PlaceholderColor3 = T.SubText, TextColor3 = T.Text, Font = T.Font, TextSize = 12,
					ClearTextOnFocus = o.ClearOnFocus or false, Parent = row,
				})
				Corner(box, 4)
				local st = Stroke(box, T.Stroke, 1, 0.8)
				box.Focused:Connect(function() Tween(st, { Transparency = 0 }, 0.12) end)

				local obj = { Value = box.Text }
				function obj:Set(t, silent)
					box.Text = t; self.Value = t
					if o.Flag then Nova.Flags[o.Flag] = t end
					if not silent then safe(o.Callback, t, false) end
				end
				function obj:Get() return self.Value end
				box.FocusLost:Connect(function(enter)
					Tween(st, { Transparency = 0.8 }, 0.12)
					obj.Value = box.Text
					if o.Flag then Nova.Flags[o.Flag] = box.Text end
					safe(o.Callback, box.Text, enter)
				end)
				if o.Flag then Nova.Flags[o.Flag] = box.Text end
				return reg(o, obj)
			end

			-- Keybind -------------------------------------------------
			function Section:AddKeybind(o)
				o = o or {}
				local row = Row(32)
				Lbl({
					Size = UDim2.new(1, -90, 1, 0), Position = UDim2.fromOffset(10, 0),
					Text = o.Name or "Keybind", Parent = row,
				})
				local btn = New("TextButton", {
					Size = UDim2.fromOffset(70, 22), AnchorPoint = Vector2.new(1, 0.5),
					Position = UDim2.new(1, -6, 0.5, 0), BackgroundColor3 = T.Panel,
					Text = "None", TextColor3 = T.SubText, Font = T.Font, TextSize = 12,
					AutoButtonColor = false, Parent = row,
				})
				Corner(btn, 4)
				Stroke(btn, T.Stroke, 1, 0.8)

				local obj = { Value = nil }
				local listening = false
				function obj:Set(key, silent)
					self.Value = key
					btn.Text = key and key.Name or "None"
					btn.TextColor3 = key and T.Text or T.SubText
					if o.Flag then Nova.Flags[o.Flag] = key end
					if not silent then safe(o.OnChanged, key) end
				end
				function obj:Get() return self.Value end

				btn.MouseButton1Click:Connect(function()
					listening = true
					btn.Text = "..."
					btn.TextColor3 = T.Accent
				end)
				bind(UserInputService.InputBegan, function(i, gp)
					if listening then
						if i.UserInputType == Enum.UserInputType.Keyboard then
							listening = false
							obj:Set(i.KeyCode ~= Enum.KeyCode.Escape and i.KeyCode or nil)
						end
						return
					end
					if not gp and obj.Value and i.KeyCode == obj.Value then safe(o.Callback) end
				end)
				obj:Set(o.Default, true)
				return reg(o, obj)
			end

			return Section
		end

		return Tab
	end

	return Window
